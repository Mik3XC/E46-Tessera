cmake_minimum_required(VERSION 3.21)
project(E46Tessera LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(E46_BUILD_TOOLS "Build the small cryptographic diagnostic CLIs" OFF)
option(E46_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)

if(NOT UNIX)
  message(FATAL_ERROR "This experimental node and wallet currently require POSIX APIs.")
endif()
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("int main() { __int128 x = 1; return int(x >> 64); }" E46_HAS_INT128)
if(NOT E46_HAS_INT128)
  message(FATAL_ERROR "E46 requires a C++20 compiler with __int128 support (Clang or GCC).")
endif()

# Preserve the upstream provider pin and built-in seeded-keygen self-check.
if(NOT OPENSSL_ROOT_DIR AND EXISTS /opt/homebrew/opt/openssl@3)
  set(OPENSSL_ROOT_DIR /opt/homebrew/opt/openssl@3)
endif()
find_package(OpenSSL 3.6.5 EXACT REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)

add_library(e46_options INTERFACE)
target_compile_options(e46_options INTERFACE -Wall -Wextra -Werror)
if(E46_SANITIZE)
  target_compile_options(e46_options INTERFACE
    -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(e46_options INTERFACE -fsanitize=address,undefined)
endif()

function(e46_library name)
  add_library(${name} STATIC src/${name}.cpp)
  add_library(E46::${name} ALIAS ${name})
  target_include_directories(${name} PUBLIC include)
  target_link_libraries(${name} PUBLIC e46_options ${ARGN})
endfunction()

e46_library(rl)
e46_library(shake OpenSSL::Crypto)
e46_library(bech32m)
e46_library(merkle shake)
e46_library(rad rl merkle)
e46_library(addresses rl shake bech32m rad)
e46_library(header rl)
e46_library(emission)
e46_library(keys shake OpenSSL::Crypto)
e46_library(transaction addresses merkle keys)
e46_library(block_qtl transaction rad merkle shake Threads::Threads)
e46_library(genesis transaction header addresses merkle Threads::Threads)
e46_library(qtl_v2 shake OpenSSL::Crypto)
e46_library(node genesis transaction header emission block_qtl addresses merkle rl shake Threads::Threads)
target_sources(node PRIVATE src/checkpoints.cpp src/p2p.cpp)

function(e46_tool name source)
  add_executable(${name} tools/${source})
  target_include_directories(${name} PRIVATE tools src)
  target_link_libraries(${name} PRIVATE e46_options ${ARGN})
endfunction()

e46_tool(e46-node e46_node.cpp node)
e46_tool(e46-wallet e46_wallet.cpp transaction keys addresses qtl_v2 OpenSSL::Crypto)
e46_tool(addresses_cli addresses_cli.cpp addresses)
e46_tool(transaction_cli transaction_cli.cpp transaction)

if(E46_BUILD_TOOLS)
  foreach(module IN ITEMS rl shake bech32m merkle rad header emission keys block_qtl genesis qtl_v2)
    e46_tool(${module}_cli ${module}_cli.cpp ${module})
  endforeach()
endif()

  add_executable(e46-storage-checks tests/storage_checks.cpp)
